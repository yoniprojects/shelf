  // Resume where I left off: if the app was freshly launched (e.g. from a
  // home-screen icon/bookmark, no referrer) and the last page visited wasn't
  // the dashboard, jump straight there before anything renders.
  // Exceptions: an explicit "back to shelf" link (?home=1) means the user is
  // deliberately choosing the dashboard; and any navigation that has a
  // same-site referrer means the user just clicked a menu link on purpose,
  // so it should never be overridden.
  (function(){
    try{
      if (new URLSearchParams(location.search).has('home')) return;
      if (document.referrer && document.referrer.indexOf(location.origin) === 0) return;
      // Only ever redirect once per browser session (true fresh launch).
      // Without this, some mobile browsers/PWAs report no referrer on an
      // ordinary reload too, which could bounce you back to a stale
      // sub-app page (or the dashboard) every time you're just navigating
      // around inside the app.
      if (sessionStorage.getItem('shelf:sessionStarted')) return;
      sessionStorage.setItem('shelf:sessionStarted', '1');
      var last = localStorage.getItem('shelf:lastPage');
      if (last && last !== 'index.html') {
        location.replace(last);
      }
    } catch(e){ /* storage unavailable — just load normally */ }
  })();

function sortSuggestionsByInput(list, typedValue) {
  const query = (typedValue || "").trim().toLowerCase();
  if (!query) return [...list].sort((a, b) => a.localeCompare(b));
  
  return [...list].sort((a, b) => {
    const aLower = a.toLowerCase();
    const bLower = b.toLowerCase();
    const aIdx = aLower.indexOf(query);
    const bIdx = bLower.indexOf(query);

    // If both contain the query string, sort by the earliest occurrence index of that string
    if (aIdx !== -1 && bIdx !== -1) {
      if (aIdx !== bIdx) return aIdx - bIdx;
    } else if (aIdx !== -1) {
      return -1;
    } else if (bIdx !== -1) {
      return 1;
    }

    // Fallback: check matching order of the individual letters typed
    let aMatchIndex = 0;
    let bMatchIndex = 0;
    for (let char of query) {
      aMatchIndex = aLower.indexOf(char, aMatchIndex);
      bMatchIndex = bLower.indexOf(char, bMatchIndex);
      if (aMatchIndex === -1) aMatchIndex = Infinity;
      if (bMatchIndex === -1) bMatchIndex = Infinity;
    }

    if (aMatchIndex !== bMatchIndex) {
      return aMatchIndex - bMatchIndex;
    }

    return a.localeCompare(b);
  });
}

  const isIOS = /iP(hone|ad|od)/.test(navigator.userAgent) ||
    (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1);

  if (!isIOS) {
    const url = URL.createObjectURL(blob);
    const dlAnchor = document.createElement('a');
    dlAnchor.href = url;
    dlAnchor.download = filename;
    document.body.appendChild(dlAnchor);
    dlAnchor.click();
    dlAnchor.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  } else {
    // iOS fallback: open in a new tab. The user can then use the Share
    // button in Safari to save/send the file.
    const url = URL.createObjectURL(blob);
    const win = window.open(url, '_blank');
    if (!win) {
      // Popup blocked — fall back to navigating the current tab.
      window.location.href = url;
    }
    setTimeout(() => URL.revokeObjectURL(url), 60000);
  }

  state.modal = null;
  renderModal();
}

      try {
        const parsed = JSON.parse(e.target.result);

        if (parsed && parsed.version === 2 && Array.isArray(parsed.profiles)) {
          // Full multi-profile, multi-app backup produced by the current export.
          for (const profileEntry of parsed.profiles) {
            await restoreFullProfileBackup(profileEntry);

          location.reload();
          return;

        // current profile's books/wishlist (and possibly goals).

function resizeImageFile(file, maxDim, cb){
  const reader = new FileReader();
  reader.onload = (e)=>{
    const img = new Image();
    img.onload = ()=>{
      let w = img.width, h = img.height;
      if(w > h){ if(w > maxDim){ h = Math.round(h*maxDim/w); w = maxDim; } }
      else { if(h > maxDim){ w = Math.round(w*maxDim/h); h = maxDim; } }
      const canvas = document.createElement("canvas");
      canvas.width = w; canvas.height = h;
      canvas.getContext("2d").drawImage(img, 0, 0, w, h);
      cb(canvas.toDataURL("image/jpeg", 0.78));
    };
    img.onerror = ()=>cb(null);
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}

function computeFunFacts(books){
  const facts = [];
  const monthNames = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  const completed = books.filter(b=>b.status==="done" && b.dateFinished);
  const now = new Date();
  const cutoff = new Date(now); cutoff.setDate(cutoff.getDate()-30);

  const recent = completed.filter(b=>{ const d=new Date(b.dateFinished+"T00:00:00"); return !isNaN(d) && d>=cutoff && d<=now; });
  if(recent.length){
    const gmap={}; recent.forEach(b=>{ if(b.genre) gmap[b.genre]=(gmap[b.genre]||0)+1; });
    const top = Object.entries(gmap).sort((a,b)=>b[1]-a[1])[0];
    if(top) facts.push({icon:"📚", text:`In the past month you've read mostly <strong>${esc(top[0])}</strong> (${top[1]} of ${recent.length} book${recent.length===1?"":"s"} finished).`});
    else facts.push({icon:"📚", text:`You finished ${recent.length} book${recent.length===1?"":"s"} in the past month — nice pace!`});
  }

  const rated = completed.filter(b=>b.rating>0).sort((a,b)=>b.rating-a.rating || (b.dateFinished||"").localeCompare(a.dateFinished||""))[0];
  if(rated) facts.push({icon:"⭐", text:`Your top-rated read is <strong>${esc(rated.title)}</strong> at ${rated.rating}★.`});

  const withPages = completed.filter(b=>b.pagesHours && !isNaN(Number(b.pagesHours))).sort((a,b)=>Number(b.pagesHours)-Number(a.pagesHours))[0];
  if(withPages) facts.push({icon:"📖", text:`Your longest read was <strong>${esc(withPages.title)}</strong> at ${withPages.pagesHours} pages.`});

  const monthMap={};
  completed.forEach(b=>{ const key=b.dateFinished.slice(0,7); monthMap[key]=(monthMap[key]||0)+1; });
  const topMonth = Object.entries(monthMap).sort((a,b)=>b[1]-a[1])[0];
  if(topMonth && topMonth[1]>1){
    const [y,m] = topMonth[0].split("-");
    facts.push({icon:"🔥", text:`Your best month was ${monthNames[Number(m)-1]} ${y} — ${topMonth[1]} books finished.`});
  }

  const ratedAll = completed.filter(b=>b.rating>0);
  if(ratedAll.length){
    const avg = (ratedAll.reduce((s,b)=>s+b.rating,0)/ratedAll.length).toFixed(1);
    facts.push({icon:"🌟", text:`Your average rating for finished books is ${avg}★ across ${ratedAll.length} rated read${ratedAll.length===1?"":"s"}.`});
  }

  const dnfCount = books.filter(b=>b.status==="dnf").length;
  if(dnfCount) facts.push({icon:"👋", text:`You've set aside ${dnfCount} book${dnfCount===1?"":"s"} as did-not-finish — good riddance!`});

  const gmapAll={}; completed.forEach(b=>{ if(b.genre) gmapAll[b.genre]=(gmapAll[b.genre]||0)+1; });
  const topAll = Object.entries(gmapAll).sort((a,b)=>b[1]-a[1])[0];
  if(topAll) facts.push({icon:"🏆", text:`All-time, your most-read genre is <strong>${esc(topAll[0])}</strong> with ${topAll[1]} book${topAll[1]===1?"":"s"} finished.`});

  return facts;
}

let goodreadsParsed = null;
function parseCSV(text){
  const rows = [];
  let row = [], field = "", inQuotes = false;
  for(let i=0;i<text.length;i++){
    const c = text[i];
    if(inQuotes){
      if(c === '"'){ if(text[i+1] === '"'){ field += '"'; i++; } else inQuotes = false; }
      else field += c;
    } else {
      if(c === '"') inQuotes = true;
      else if(c === ","){ row.push(field); field = ""; }
      else if(c === "\n"){ row.push(field); rows.push(row); row = []; field = ""; }
      else if(c === "\r"){ /* skip */ }
      else field += c;
    }
  }
  if(field.length || row.length){ row.push(field); rows.push(row); }
  return rows.filter(r => r.length > 1 || (r.length === 1 && r[0] !== ""));
}
function normalizeGoodreadsDate(s){
  if(!s) return "";
  const m = s.trim().match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
  if(!m) return "";
  return `${m[1]}-${m[2].padStart(2,"0")}-${m[3].padStart(2,"0")}`;
}
function mapGoodreadsFormat(binding){
  const s = (binding||"").toLowerCase();
  if(s.includes("audio")) return "Audio";
  if(s.includes("kindle")||s.includes("ebook")||s.includes("digital")||s.includes("nook")) return "Digital";
  return "Physical";
}
function mapGoodreadsStatus(shelf){
  const s = (shelf||"").toLowerCase();
  if(s.includes("currently")) return "reading";
  if(s.includes("to-read")) return "want";
  return "done";
}
function rowsToBooks(rows){
  if(rows.length < 2) return [];
  const header = rows[0].map(h=>h.trim());
  const idx = name => header.indexOf(name);
  const iTitle=idx("Title"), iAuthor=idx("Author"), iRating=idx("My Rating"), iDateRead=idx("Date Read"),
        iShelf=idx("Exclusive Shelf"), iBinding=idx("Binding"), iPages=idx("Number of Pages"),
        iReview=idx("My Review"), iNotes=idx("Private Notes"), iShelves=idx("Bookshelves");
  const out = [];
  for(let r=1;r<rows.length;r++){
    const row = rows[r];
    if(!row || row.length<2) continue;
    const title = iTitle>=0 ? row[iTitle].trim() : "";
    if(!title) continue;
    const shelvesRaw = iShelves>=0 ? row[iShelves] : "";
    const genreGuess = shelvesRaw ? (shelvesRaw.split(",").map(s=>s.trim()).filter(Boolean)[0]||"") : "";
    const status = mapGoodreadsStatus(iShelf>=0?row[iShelf]:"");
    out.push({
      id: uid(), isExample:false, title,
      author: iAuthor>=0 ? row[iAuthor].trim() : "",
      genre: genreGuess, subGenre: "",
      format: mapGoodreadsFormat(iBinding>=0?row[iBinding]:""),
      source: "", status: status,
      priority: (status === "done" || status === "dnf") ? "" : "Interested",
      rating: Number(iRating>=0?row[iRating]:0)||0,
      dateFinished: normalizeGoodreadsDate(iDateRead>=0?row[iDateRead]:""),
      language: "English", pagesHours: iPages>=0 ? row[iPages].trim() : "",
      review: iReview>=0 ? row[iReview].trim() : "", notes: iNotes>=0 ? row[iNotes].trim() : "",
      recommendedBy: "", coverUrl: "",
    });
  }
  return out;
}
function handleGoodreadsFile(file){
  if(!file) return;
  const reader = new FileReader();
  reader.onload = (e)=>{
    try{ goodreadsParsed = rowsToBooks(parseCSV(e.target.result)); }
    catch(err){ goodreadsParsed = []; }
    renderModal();
  };
  reader.readAsText(file);
}
function runGoodreadsImport(){
  if(!goodreadsParsed || goodreadsParsed.length===0) return;
  const skipDupes = document.getElementById("gr-skip-dupes") ? document.getElementById("gr-skip-dupes").checked : true;
  const existingKeys = new Set(state.books.map(b=>(b.title+"|"+b.author).toLowerCase()));
  goodreadsParsed.forEach(b=>{
    const key = (b.title+"|"+b.author).toLowerCase();
    if(skipDupes && existingKeys.has(key)) return;
    state.books.push(b);
    existingKeys.add(key);
  });
  persistBooks();
  goodreadsParsed = null;
  state.modal = null;
  state.tab = "tracker";
  lsSet('shelf:lastTab', state.tab);
  render();
}

function allSeriesNames(){
  const names = new Set();
  state.books.forEach(b=>{ if(b.seriesName) names.add(b.seriesName); });
  state.wishlist.forEach(w=>{ if(w.seriesName) names.add(w.seriesName); });
  return Array.from(names);
}
function findSeriesEntries(seriesName){
  const target = (seriesName||"").trim().toLowerCase();
  if(!target) return [];
  const fromBooks = state.books.filter(b=>(b.seriesName||"").trim().toLowerCase()===target)
    .map(b=>({kind:"book", id:b.id, title:b.title, author:b.author, number:b.seriesNumber||"", status:b.status}));
  const fromWish = state.wishlist.filter(w=>(w.seriesName||"").trim().toLowerCase()===target)
    .map(w=>({kind:"wish", id:w.id, title:w.title, author:w.author, number:w.seriesNumber||""}));
  return [...fromBooks, ...fromWish].sort((a,b)=>{
    const na = parseFloat(a.number), nb = parseFloat(b.number);
    const va = isNaN(na) ? Infinity : na, vb = isNaN(nb) ? Infinity : nb;
    if(va!==vb) return va-vb;
    return (a.title||"").localeCompare(b.title||"");
  });
}

    const url = "https://www.googleapis.com/books/v1/volumes?maxResults=8&q=" + encodeURIComponent(q) + (key ? "&key="+encodeURIComponent(key) : "");
    const res = await fetch(url);
    if(seq !== gbSearchSeq) return; // a newer search superseded this one
    if(!res.ok){
      let reason;
      if(res.status===429) reason = "Google is rate-limiting search traffic right now — try again in a moment.";
      else if(res.status===403) reason = "Google rejected the request (403).";
      else reason = "Request failed (status " + res.status + ").";
      box.innerHTML = `<div style="padding:8px 2px;font-size:12px;color:#A2572B;">${esc(reason)}</div>`;
      return;
    }
    const data = await res.json();
    const items = (data.items||[]).filter(it=>it.volumeInfo && it.volumeInfo.title);
    gbResults = items;
    if(!items.length){
      box.innerHTML = `<div style="padding:8px 2px;font-size:12px;color:var(--muted);">No matches found. Try a different search, or fill the form in by hand.</div>`;
      return;
    }
    box.innerHTML = items.map((it,i)=>{
      const v = it.volumeInfo;
      const thumb = v.imageLinks && (v.imageLinks.thumbnail || v.imageLinks.smallThumbnail);
      const authors = (v.authors||[]).join(", ");
      const year = v.publishedDate ? v.publishedDate.slice(0,4) : "";
      return `
        <button type="button" class="gb-result" data-action="gb-pick" data-index="${i}">
          <div class="gb-result-cover">${thumb?`<img src="${esc(thumb.replace('http://','https://'))}" alt="" />`:"📖"}</div>
          <div class="gb-result-text">
            <div class="gb-result-title">${esc(v.title)}</div>
            <div class="gb-result-meta">${esc(authors)}${year?` · ${year}`:""}</div>
          </div>
        </button>`;
    }).join("");
  }catch(err){
    if(seq !== gbSearchSeq) return;
    box.innerHTML = `<div style="padding:8px 2px;font-size:12px;color:#A2572B;">Search failed. Check your internet connection and try again.</div>`;
  }
}

function getFavKeys(){

function tabBtnHTML(t){

function linkBtnHTML(l){

function navItemHTML(key, innerHtml){
  const isFav = getFavKeys().includes(key);
  return `<div class="nav-item">
    ${innerHtml}
    <button type="button" class="nav-fav-btn${isFav?" is-fav":""}" data-action="toggle-fav" data-id="${esc(key)}" title="${isFav?"Remove from favorites":"Add to favorites"}" aria-label="Favorite">${isFav?"★":"☆"}</button>
  </div>`;

// Renders a group of nav entries (tabs/links) as favorites-first, then the
// rest, with a dotted separator between the two groups when both are non-empty.
function navGroupHTML(tabs, links){
  const favKeys = getFavKeys();
  const allEntries = [...tabs.map(t=>({...t,type:"tab"})), ...links.map(l=>({...l,type:"link"}))];
  const favEntries = favKeys.map(k=>allEntries.find(e=>e.key===k)).filter(Boolean);
  const restTabs = tabs.map(t=>({...t,type:"tab"})).filter(t=>!favKeys.includes(t.key));
  const restLinks = links.map(l=>({...l,type:"link"})).filter(l=>!favKeys.includes(l.key));
  const entryHTML = (e)=> navItemHTML(e.key, e.type==="tab" ? tabBtnHTML(e) : linkBtnHTML(e));
  return `
    ${favEntries.map(entryHTML).join("")}
    ${favEntries.length ? '<hr class="nav-separator" />' : ""}
    ${restTabs.map(entryHTML).join("")}
    ${(restTabs.length && restLinks.length) ? '<hr class="nav-separator" />' : ""}
    ${restLinks.map(entryHTML).join("")}
  `;
}

/* ---------------- nav favorite star: long-press reveal (touch) ---------------- */
// Desktop reveals the star on :hover (pure CSS). Touch devices have no hover,
// so a ~500ms press-and-hold on a nav item reveals its star instead; lifting
// the finger after a completed long-press suppresses the click that would
// otherwise navigate/switch tabs, so the user can then tap the star.
(function(){
  let holdTimer = null;
  let holdFired = false;
  document.addEventListener("touchstart", (e)=>{
    const item = e.target.closest(".nav-item");
    holdFired = false;
    if(!item) return;
    holdTimer = setTimeout(()=>{
      item.classList.add("show-fav");
      holdFired = true;
      if(navigator.vibrate) navigator.vibrate(15);
    }, 500);
  }, {passive:true});
  document.addEventListener("touchend", (e)=>{
    if(holdTimer){ clearTimeout(holdTimer); holdTimer = null; }
    if(holdFired){ e.preventDefault(); holdFired = false; }
  }, {passive:false});
  document.addEventListener("touchmove", ()=>{
    if(holdTimer){ clearTimeout(holdTimer); holdTimer = null; }
  }, {passive:true});
  // Tapping/clicking elsewhere hides any revealed stars again.
  document.addEventListener("click", (e)=>{
    if(e.target.closest(".nav-item")) return;
    document.querySelectorAll(".nav-item.show-fav").forEach(el=>el.classList.remove("show-fav"));
  });
})();

/* ---------------- dashboard: top reads dynamic fit ---------------- */
function fitTopReads(){
  const row = document.getElementById("top-reads-row");
  if(!row) return;
  const items = Array.from(row.querySelectorAll(".top-read-item"));
  if(!items.length) return;
  items.forEach(it=>{ it.style.display = ""; });
  const rowWidth = row.clientWidth;
  const itemWidth = items[0].getBoundingClientRect().width;
  if(itemWidth <= 0) return;
  const gapPx = parseFloat(getComputedStyle(row).columnGap || getComputedStyle(row).gap) || 0;
  let count = Math.floor((rowWidth + gapPx) / (itemWidth + gapPx));
  count = Math.max(1, Math.min(items.length, count));
  items.forEach((it,i)=>{ it.style.display = i < count ? "" : "none"; });
}

/* ---------------- dashboard: fun-facts dynamic sizing ---------------- */
function fitFunFactsHeight(){
  const row = document.getElementById("cr-ff-row");
  const crCard = document.getElementById("cr-card");
  const ffCard = document.getElementById("ff-card");
  if(!row || !crCard || !ffCard) return;
  const factList = ffCard.querySelector(".fact-list");
  if(!factList) return;

  const BASE_FONT = 14.5, BASE_GAP = 16;
  const MIN_SCALE = 0.72, MAX_SCALE = 1.65;

  // Measure natural (unstretched) heights so we can compare real content sizes,
  // not the equalized heights align-items:stretch produces.
  row.style.alignItems = "flex-start";
  factList.style.fontSize = "";
  factList.style.gap = "";

  // Only relevant when the two cards are actually sharing a row.
  const sameRow = Math.abs(crCard.getBoundingClientRect().top - ffCard.getBoundingClientRect().top) < 1;
  if(!sameRow){
    row.style.alignItems = "stretch";
    return;
  }

  const crHeight = crCard.offsetHeight;
  let ffHeight = ffCard.offsetHeight;
  if(crHeight > 0 && ffHeight > 0){
    let scale = Math.max(MIN_SCALE, Math.min(MAX_SCALE, crHeight/ffHeight));
    factList.style.fontSize = (BASE_FONT*scale).toFixed(2)+"px";
    factList.style.gap = Math.max(6, Math.round(BASE_GAP*scale))+"px";
    // One corrective pass: line-wrapping means font-size vs height isn't perfectly
    // linear, so re-measure once and nudge closer.
    ffHeight = ffCard.offsetHeight;
    if(ffHeight > 0){
      const correction = Math.max(0.85, Math.min(1.18, crHeight/ffHeight));
      scale = Math.max(MIN_SCALE, Math.min(MAX_SCALE, scale*correction));
      factList.style.fontSize = (BASE_FONT*scale).toFixed(2)+"px";
      factList.style.gap = Math.max(6, Math.round(BASE_GAP*scale))+"px";
    }
  }
  row.style.alignItems = "stretch";
}
